After watching Xiaotudui's PyTorch tutorial videos, I have gained a basic understanding of how to build a CNN model and how to write the complete training process using the CIFAR-10 dataset.

In this experiment, I would like to compare the performance of the CNN model before and after adding the ReLU activation function to better understand how ReLU affects the model's learning ability and performance.

I defined two cnn models, Tudui_No_ReLU and Tudui_ReLU, let's explore what will happen.

import torch
import torchvision
from torch import nn
from torch.utils.data import DataLoader

#download data
train_data = torchvision.datasets.CIFAR10(root="./data",train=True,transform=torchvision.transforms.ToTensor(),
                                          download=True)
test_data = torchvision.datasets.CIFAR10(root="./data",train=False,transform=torchvision.transforms.ToTensor(),
                                         download=True)

#Dataloader
train_dataloader = DataLoader(train_data,batch_size=64,shuffle=True)
test_dataloader = DataLoader(test_data,batch_size=64,shuffle=False)

train_data_size = len(train_data)
test_data_size = len(test_data)


#define class Tudui_No_ReLU
class Tudui_No_ReLU(nn.Module):
    def __init__(self):
        super(Tudui_No_ReLU,self).__init__()
        self.model = nn.Sequential(
            nn.Conv2d(3,32,5,1,2),
            nn.MaxPool2d(2),
            nn.Conv2d(32,32,5,1,2),
            nn.MaxPool2d(2),
            nn.Conv2d(32,64,5,1,2),
            nn.MaxPool2d(2),
            nn.Flatten(),
            nn.Linear(1024,64),
            nn.Linear(64,10)
        )

    def forward(self,x):
        x = self.model(x)
        return x

#define tudui and other parameters
tudui = Tudui_No_ReLU()
loss_fn = nn.CrossEntropyLoss()
optimizer = torch.optim.SGD(tudui.parameters(),lr=1e-2)

epoch = 5
no_relu_test_loss = []
no_relu_test_accuracy = []


#train the model
for i in range(epoch):
    print("=====================epoch{}===========================".format(i))
    tudui.train()
    total_test_loss = 0
    test_accuracy = 0

    for data in train_dataloader:
        imgs, targets = data
        output = tudui(imgs)
        loss_train = loss_fn(output,targets)

        optimizer.zero_grad()
        loss_train.backward()
        optimizer.step()  

    tudui.eval()
    with torch.no_grad():
        for data in test_dataloader:
            imgs, targets = data
            output = tudui(imgs)
            loss_test = loss_fn(output,targets)
            total_test_loss = total_test_loss + loss_test.item()
            test_accuracy = test_accuracy + (output.argmax(1) == targets).sum()

    print("Epoch{}, Total test loss without ReLu:{}".format(i,total_test_loss))
    print("Epoch{}, Test accuracy without ReLu:{}".format(i,test_accuracy/test_data_size))

    #record the results
    no_relu_test_loss.append(total_test_loss)
    no_relu_test_accuracy.append(test_accuracy/test_data_size)

From the figures, we can see that when the number of epochs is 5 and the CNN contains three convolutional layers, adding ReLU activation functions does not significantly improve the accuracy.

This may be because the model is relatively simple and the training time is short, so the advantage of introducing non-linearity is not obvious in this experiment. However, this does not mean that ReLU is unnecessary. In deeper or more complex CNNs, activation functions such as ReLU are important because they allow the model to learn non-linear and more complex features.

Therefore, whether and how to use activation functions depends on the model architecture, dataset, and training settings.
