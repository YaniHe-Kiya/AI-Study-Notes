ViT Attention Visualization¶

This experiment visualizes the attention learned by a Vision Transformer by examining the CLS token's attention to image patches. It first averages the attention weights across all attention heads to generate an overall attention map, then visualizes each attention head separately to investigate whether different heads focus on different regions of the image.

import torch
from torch import nn
import math

class MultiHeadSelfAttention(nn.Module):

    def __init__(self, embed_dim=128, num_heads=4):
        super().__init__()

        assert embed_dim % num_heads == 0

        self.embed_dim = embed_dim
        self.num_heads = num_heads
        self.head_dim = embed_dim // num_heads

        self.q_proj = nn.Linear(embed_dim, embed_dim)
        self.k_proj = nn.Linear(embed_dim, embed_dim)
        self.v_proj = nn.Linear(embed_dim, embed_dim)

        self.out_proj = nn.Linear(embed_dim, embed_dim)

        # save attention
        self.attention_weights = None

    def forward(self, x):

        B, N, D = x.shape

        Q = self.q_proj(x)
        K = self.k_proj(x)
        V = self.v_proj(x)

        Q = Q.reshape(B, N, self.num_heads, self.head_dim).transpose(1, 2)
        K = K.reshape(B, N, self.num_heads, self.head_dim).transpose(1, 2)
        V = V.reshape(B, N, self.num_heads, self.head_dim).transpose(1, 2)

        scores = Q @ K.transpose(-2, -1)
        scores = scores / math.sqrt(self.head_dim)

        attention = torch.softmax(scores, dim=-1)

        # save attention
        self.attention_weights = attention.detach()

        out = attention @ V
        out = out.transpose(1, 2)
        out = out.reshape(B, N, self.embed_dim)

        out = self.out_proj(out)

        return out

class TransformerBlock(nn.Module):

    def __init__(
        self,
        embed_dim=128,
        num_heads=4,
        mlp_dim=512
    ):
        super().__init__()

        self.norm1 = nn.LayerNorm(embed_dim)

        self.attention = MultiHeadSelfAttention(
            embed_dim=embed_dim,
            num_heads=num_heads
        )

        self.norm2 = nn.LayerNorm(embed_dim)

        self.mlp = nn.Sequential(
            nn.Linear(embed_dim, mlp_dim),
            nn.GELU(),
            nn.Linear(mlp_dim, embed_dim)
        )

    def forward(self, x):

        # Attention + Residual
        x = x + self.attention(
            self.norm1(x)
        )

        # MLP + Residual
        x = x + self.mlp(
            self.norm2(x)
        )

        return x


class TransformerEncoder(nn.Module):

    def __init__(
        self,
        depth=4,
        embed_dim=128,
        num_heads=4,
        mlp_dim=512
    ):
        super().__init__()

        self.blocks = nn.ModuleList([
            TransformerBlock(
                embed_dim=embed_dim,
                num_heads=num_heads,
                mlp_dim=mlp_dim
            )
            for _ in range(depth)
        ])

    def forward(self, x):

        for block in self.blocks:
            x = block(x)

        return x


class MiniViT(nn.Module):

    def __init__(
        self,
        image_size=32,
        patch_size=4,
        in_channels=3,
        num_classes=10,
        embed_dim=128,
        depth=4,
        num_heads=4,
        mlp_dim=512
    ):
        super().__init__()
        num_patches = (
            image_size // patch_size
        ) ** 2
        self.patch_embedding = nn.Conv2d(
            in_channels=in_channels,
            out_channels=embed_dim,
            kernel_size=patch_size,
            stride=patch_size
        )
        self.cls_token = nn.Parameter(
            torch.randn(
                1,
                1,
                embed_dim
            )
        )
        self.position_embedding = nn.Parameter(
            torch.randn(
                1,
                num_patches + 1,
                embed_dim
            )
        )
        self.encoder = TransformerEncoder(
            depth=depth,
            embed_dim=embed_dim,
            num_heads=num_heads,
            mlp_dim=mlp_dim
        )
        self.norm = nn.LayerNorm(
            embed_dim
        )
        self.classifier = nn.Linear(
            embed_dim,
            num_classes
        )

    def forward(self, x):

        B = x.shape[0]
        x = self.patch_embedding(x)
        x = x.flatten(2)
        x = x.transpose(1, 2)
        cls_tokens = self.cls_token.expand(
            B,
            -1,
            -1
        )
        x = torch.cat(
            [cls_tokens, x],
            dim=1
        )
        x = x + self.position_embedding
        x = self.encoder(x)
        x = self.norm(x)
        x = x[:, 0, :]
        x = self.classifier(x)
        return x

import torchvision
from torch.utils.data import DataLoader
train_dataset = torchvision.datasets.CIFAR10("../data", train = True, transform=torchvision.transforms.ToTensor(),
                                             download= True)

test_dataset = torchvision.datasets.CIFAR10("../data", train=False, transform=torchvision.transforms.ToTensor(),
                                            download = True)

train_dataloader = DataLoader(train_dataset,batch_size=8,shuffle=True)
test_dataloader = DataLoader(test_dataset,batch_size = 8, shuffle = False)

epoch = 5
loss_fn = torch.nn.CrossEntropyLoss()
mini_vit = MiniViT()

optimizer = torch.optim.SGD(mini_vit.parameters(),lr=1e-2)

for i in range(epoch):
    total_loss_train = 0 
    mini_vit.train()
    print("=======Epoch{}=======".format(i))
    for data in train_dataloader:
        images, targets = data
        optimizer.zero_grad()
        output = mini_vit(images)
        loss_train = loss_fn(output,targets)
        loss_train.backward()
        optimizer.step()
        total_loss_train +=  loss_train.item()
    print("Total Train Loss:",total_loss_train)
    print("Average Train Loss:", total_loss_train / len(train_dataloader))

    with torch.no_grad():
        mini_vit.eval()
        total_accuracy = 0
        total_loss_test = 0
        for data in test_dataloader:
            images, targets = data
            output = mini_vit(images)
            loss_test = loss_fn(output,targets)
            total_loss_test += loss_test.item()
            total_accuracy += (torch.argmax(output,dim=1) == targets).sum().item()

        print("Test accuracy:", total_accuracy/len(test_dataset))
        print("Average Test Loss:", total_loss_test/len(test_dataloader))

import torch
import torch.nn.functional as F
import matplotlib.pyplot as plt


def normalize_map(attn_map):
    attn_map = attn_map - attn_map.min()

    if attn_map.max() > 0:
        attn_map = attn_map / attn_map.max()

    return attn_map


def visualize_average_attention(
    model,
    dataset,
    index=0,
    patch_size=4,
    class_names=None
):
    model.eval()

    # 1. One Image
    image, label = dataset[index]
    input_image = image.unsqueeze(0)

    # 2. Forward
    with torch.no_grad():
        output = model(input_image)

    pred = torch.argmax(output, dim=1).item()

    # 3. Get attention from the last Transformer Block
    attention = (
        model.encoder
        .blocks[-1]
        .attention
        .attention_weights
    )

    print("Attention shape:", attention.shape)

    # [1, num_heads, 65, 65]
    # Get CLS attention to the 64 patch tokens
    cls_attention = attention[0, :, 0, 1:]

    print("CLS attention:", cls_attention.shape)

    # 4. Average attention across all heads
    avg_attention = cls_attention.mean(dim=0)

    # 5. Reshape back to the patch grid
    H = image.shape[1]
    W = image.shape[2]

    grid_h = H // patch_size
    grid_w = W // patch_size

    avg_attention = avg_attention.reshape(
        grid_h,
        grid_w
    )

    # 6. Resize the attention map to the original image size
    avg_attention = avg_attention.unsqueeze(0).unsqueeze(0)

    avg_attention = F.interpolate(
        avg_attention,
        size=(H, W),
        mode="bilinear",
        align_corners=False
    )

    avg_attention = avg_attention.squeeze()
    avg_attention = normalize_map(avg_attention)

    # 7. Convert image format for visualization
    image_show = image.permute(1, 2, 0)

    if class_names is not None:
        true_name = class_names[label]
        pred_name = class_names[pred]
        title_info = f"True: {true_name} | Pred: {pred_name}"
    else:
        title_info = f"True: {label} | Pred: {pred}"

    # 8. Visualization
    plt.figure(figsize=(10, 4))

    plt.subplot(1, 2, 1)
    plt.imshow(image_show,interpolation="bicubic")
    plt.title("Original Image")
    plt.axis("off")

    plt.subplot(1, 2, 2)
    plt.imshow(image_show,interpolation="bicubic")
    plt.imshow(
        avg_attention,
        cmap="jet",
        alpha=0.5
    )
    plt.title("Average Attention")
    plt.axis("off")

    plt.suptitle(title_info)

    plt.tight_layout()
    plt.show()

def visualize_individual_heads(
    model,
    dataset,
    index=0,
    patch_size=4,
    class_names=None
):
    model.eval()

    # 1. Get one image
    image, label = dataset[index]
    input_image = image.unsqueeze(0)

    # 2. Forward
    with torch.no_grad():
        output = model(input_image)

    pred = torch.argmax(output, dim=1).item()

    # 3. Attention
    attention = (
        model.encoder
        .blocks[-1]
        .attention
        .attention_weights
    )

    # CLS -> patches
    cls_attention = attention[0, :, 0, 1:]

    num_heads = cls_attention.shape[0]

    # 4. Patch grid
    H = image.shape[1]
    W = image.shape[2]

    grid_h = H // patch_size
    grid_w = W // patch_size

    head_maps = []

    # 5. Process each attention head separately
    for h in range(num_heads):

        head_attention = cls_attention[h]

        head_attention = head_attention.reshape(
            grid_h,
            grid_w
        )

        head_attention = (
            head_attention
            .unsqueeze(0)
            .unsqueeze(0)
        )

        head_attention = F.interpolate(
            head_attention,
            size=(H, W),
            mode="bilinear",
            align_corners=False
        )

        head_attention = head_attention.squeeze()

        head_attention = normalize_map(
            head_attention
        )

        head_maps.append(
            head_attention
        )

    # 6. Image
    image_show = image.permute(1, 2, 0)

    if class_names is not None:
        true_name = class_names[label]
        pred_name = class_names[pred]

        title_info = (
            f"True: {true_name} | "
            f"Pred: {pred_name}"
        )
    else:
        title_info = (
            f"True: {label} | "
            f"Pred: {pred}"
        )

    # 7. Visualization
    fig, axes = plt.subplots(
        1,
        num_heads,
        figsize=(4 * num_heads, 4)
    )

    for h in range(num_heads):

        axes[h].imshow(image_show)

        axes[h].imshow(
            head_maps[h],
            cmap="jet",
            alpha=0.5
        )

        axes[h].set_title(
            f"Head {h + 1}"
        )

        axes[h].axis("off")

    plt.suptitle(title_info)

    plt.tight_layout()
    plt.show()
